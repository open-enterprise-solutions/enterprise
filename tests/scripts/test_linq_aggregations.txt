// LINQ aggregations smoke test — 2026-05-11 PM
// Tests no-arg Sum/Min/Max/Average on Array and combined with GROUP BY.
//
// Expected output:
//   --- numeric array ---
//   sum=15
//   min=1
//   max=5
//   avg=3
//   --- group by country sums ---
//   FR amount sum: 100
//   UK amount sum: 50
//   USA amount sum: 75
//   --- aggregates over a filtered pipeline ---
//   filtered sum=12
//   filtered min=3
//   filtered max=5
//   filtered avg=4
//   --- aggregates with a selector ---
//   sum of doubles=30
//   max of doubles=10
//   --- aggregate over an empty pipeline ---
//   empty sum is empty: True
//
// ⚠ ONE DIALECT PER FILE. This is a BRACES script — `foreach (x in q) { }` — so its
// lambdas are `Function(x) { return ...; }` too. Three of them were written in the
// other form, `Function(x) Return ... EndFunction`, which every wholly-VES script in
// this corpus uses and which does not parse here. The compiler then reported the
// failure at the `foreach` far above, because that is where the parse gave up rather
// than where the wrong word was.

var nums = New Array;
nums.Add(1);
nums.Add(2);
nums.Add(3);
nums.Add(4);
nums.Add(5);

Message("--- numeric array ---");
Message("sum=" + nums.Sum());
Message("min=" + nums.Min());
Message("max=" + nums.Max());
Message("avg=" + nums.Average());

var orders = New Array;
orders.Add(New Structure("Country, Amount", "USA", 25));
orders.Add(New Structure("Country, Amount", "UK",  30));
orders.Add(New Structure("Country, Amount", "USA", 50));
orders.Add(New Structure("Country, Amount", "FR",  100));
orders.Add(New Structure("Country, Amount", "UK",  20));

Message("--- group by country sums ---");
var q = from o in orders
        group o.Amount by o.Country;

foreach (g in q) {
    Message(g.Key + " amount sum: " + g.Values.Sum());
}

// ---------------------------------------------------------------------------
// AGGREGATES ON A PIPELINE, not only on a materialised collection.
//
// `nums.Sum()` above works because Array declares a method of that name. Until
// Sum/Min/Max/Average joined the LINQ vocabulary, the moment a filter came
// first — `nums.Where(...).Sum()` — the receiver was a pipeline with no methods
// of its own and the answer was "Aggregate object field not found 'Sum'". The
// workaround was .ToArray().Sum(), which materialises everything a lazy
// pipeline exists to avoid.
// ---------------------------------------------------------------------------

Message("--- aggregates over a filtered pipeline ---");
var big = Function(x) { return x > 2; };
Message("filtered sum=" + nums.Where(big).Sum());           // 12  (3 + 4 + 5)
Message("filtered min=" + nums.Where(big).Min());           // 3
Message("filtered max=" + nums.Where(big).Max());           // 5
Message("filtered avg=" + nums.Where(big).Average());       // 4

Message("--- aggregates with a selector ---");
var twice = Function(x) { return x * 2; };
Message("sum of doubles=" + nums.Sum(twice));               // 30
Message("max of doubles=" + nums.Select(twice).Max());      // 10

Message("--- aggregate over an empty pipeline ---");
var none = Function(x) { return x > 100; };
Message("empty sum is empty: " + (nums.Where(none).Sum() = Undefined));   // True
